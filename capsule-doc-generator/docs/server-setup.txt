1) transfer flask server python and template directory to server
2) create virtual environment in project folder
		python3 -m venv venv
    	source venv/bin/activate
    	pip install flask gunicorn
		pip install flask_cors
3) run flask with gunicorn
    	gunicorn flask_server:app --bind 127.0.0.1:8000
	(exit venv)
4) install Nginx
    	sudo apt update
    	sudo apt install nginx
5) create Nginx config file
    	sudo nano /etc/nginx/sites-available/capsuleflask
    	(copy paste config)
6) enable config
    	sudo ln -s /etc/nginx/sites-available/capsuleflask /etc/nginx/sites-enabled/
    	sudo nginx -t  # check config
    	sudo systemctl restart nginx
7) add https (optional)
    	sudo apt install certbot python3-certbot-nginx 
    	#installs certbot
    	sudo certbot --nginx
    	#gets SSL cert and updates Nginx config
8) make flask auto-restart on boot
	sudo nano /etc/systemd/system/capsuleflask.service

	[Unit]
	Description=Gunicorn for Flask App
	After=network.target

	[Service]
	User=(linux username)
	WorkingDirectory=(directory to project folder) 
	ExecStart=(project folder)/venv/bin/gunicorn flask_server:app --bind 127.0.0.1:8000
	Restart=always

	[Install]
	WantedBy=multi-user.target
	
	sudo systemctl daemon-reload
	sudo systemctl enable capsuleflask
	sudo systemctl start capsuleflask

if restarting:
	restart app : sudo systemctl restart capsuleflask
